103: function getTrackCoords(distance: number) {
104:   const d = ((distance % L_TOTAL) + L_TOTAL) % L_TOTAL;
105:   if (d < L_STRAIGHT) {
106:     // Top straight: left to right (y = TOP_Y)
107:     return {
108:       x: LEFT_X + d,
109:       y: TOP_Y,
110:       angle: 0.0,
111:     };
112:   } else if (d < L_STRAIGHT + L_ARC) {
113:     // Right curve
114:     const arcD = d - L_STRAIGHT;
115:     const theta = -Math.PI / 2.0 + (arcD / L_ARC) * Math.PI;
116:     return {
117:       x: RIGHT_X + RADIUS * Math.cos(theta),
118:       y: (TOP_Y + BOTTOM_Y) / 2.0 + RADIUS * Math.sin(theta),
119:       angle: 90.0 + (arcD / L_ARC) * 180.0,
120:     };
121:   } else if (d < 2 * L_STRAIGHT + L_ARC) {
122:     // Bottom straight: right to left (y = BOTTOM_Y)
123:     const straightD = d - (L_STRAIGHT + L_ARC);
124:     return {
125:       x: RIGHT_X - straightD,
126:       y: BOTTOM_Y,
127:       angle: 180.0,
128:     };
129:   } else {
130:     // Left curve
131:     const arcD = d - (2 * L_STRAIGHT + L_ARC);
132:     const theta = Math.PI / 2.0 + (arcD / L_ARC) * Math.PI;
133:     return {
134:       x: LEFT_X + RADIUS * Math.cos(theta),
135:       y: (TOP_Y + BOTTOM_Y) / 2.0 + RADIUS * Math.sin(theta),
136:       angle: 270.0 + (arcD / L_ARC) * 180.0,
137:     };
138:   }
